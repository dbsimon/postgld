function jsonResp(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ---- helper: password -----------------------------------------------------

function validatePassword(pwd) {
  if (!pwd || String(pwd).trim() === '') return false;
  var stored = getConfigValue(PASSWORD_KEY) || FALLBACK_PASSWORD;
  return String(pwd) === String(stored);
}

function getConfigValue(key) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config');
    if (!sheet || sheet.getLastRow() <= 1) return null;
    var data = sheet.getDataRange().getValues();
    for (var r = 1; r < data.length; r++) {
      if (data[r][0] === key) return String(data[r][1] || '');
    }
  } catch(e) {
    Logger.log('getConfigValue error for key=' + key + ': ' + e);
  }
  return null;
}

/**
 * Ensure Config sheet has the required migration rows.
 * Safe to call on every request — only seeds missing keys.
 */
function ensureConfigDefaults() {
  var cfg = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config');
  if (!cfg) {
    cfg = SpreadsheetApp.getActiveSpreadsheet().insertSheet('Config');
  }
  var defaults = {
    'APP_PASSWORD':     ['XXXX',      'API password for web-app authentication'],
    'SCHEMA_VERSION':   ['2',         'Workbook schema version'],
    'ADMIN_PASSWORD':   ['XXXXX',  'Admin mode password (front-end)'],
    'PAGE_PASSWORD':    ['XXXX',      'Page lock password (front-end)'],
    'MIGRATION_MODE':   ['dual',      'legacy | dual | canonical'],
    'COMPAT_WARNINGS':  ['true',      'Include deprecation warnings in legacy responses'],
    'DEPRECATION_NOTICE': [DEPRECATION_MESSAGE, 'Shown to legacy clients']
  };

  // ensure headers
  if (cfg.getLastRow() === 0) {
    cfg.getRange(1, 1, 1, 4).setValues([['Key','Value','Description','Updated At']])
      .setFontWeight('bold').setBackground('#E2E8F0');
  }

  var existing = {};
  if (cfg.getLastRow() > 1) {
    var data = cfg.getRange(2, 1, cfg.getLastRow() - 1, 1).getValues();
    for (var i = 0; i < data.length; i++) { existing[data[i][0]] = true; }
  }

  var keys = Object.keys(defaults);
  for (var k = 0; k < keys.length; k++) {
    var key = keys[k];
    if (!existing[key]) {
      cfg.appendRow([key, defaults[key][0], defaults[key][1], new Date().toISOString()]);
    }
  }
}

// ---- helper: schema version -----------------------------------------------

function validateSchemaVersion(payload) {
  if (payload.schemaVersion === undefined) return '';   // optional — no error
  var v = parseInt(payload.schemaVersion, 10);
  if (isNaN(v)) return 'schemaVersion must be a number';
  if (v !== REQUIRED_SCHEMA_VERSION) return 'schemaVersion mismatch: expected ' + REQUIRED_SCHEMA_VERSION + ', got ' + v;
  return '';
}

// ---- sheet management helpers ---------------------------------------------

/**
 * Ensure a sheet exists with correct headers; create if missing.
 * Returns the sheet object.
 */
function ensureSheet(name, schema) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }

  var existing = sheet.getLastRow() > 0 ? sheet.getRange(1, 1, 1, schema.headers.length).getValues()[0] : [];
  var needsHeaders = false;
  for (var i = 0; i < schema.headers.length; i++) {
    if ((existing[i] || '') !== schema.headers[i]) { needsHeaders = true; break; }
  }
  if (needsHeaders) {
    sheet.getRange(1, 1, 1, schema.headers.length)
      .setValues([schema.headers])
      .setFontWeight('bold')
      .setBackground('#E2E8F0');
  }

  return sheet;
}

/**
 * Write data rows to a master sheet.  Clears rows 2+, writes new data, formats.
 */
function writeSheetWithHeaders(name, schema, rows) {
  if (name === 'Movements' && rows && rows.length) {
    rows = rows.map(function(row) { var copy = row.slice(); copy[6] = toSheetEffectiveDate(copy[6]); return copy; });
  }
  var sheet = ensureSheet(name, schema);

  // clear existing data (keep header)
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, schema.headers.length).clear();
  }

  if (rows && rows.length > 0) {
    sheet.getRange(2, 1, rows.length, rows[0].length)
      .setNumberFormat('@')
      .setValues(rows);
  }

  formatSheet(sheet, schema, sheet.getLastRow());
  return sheet;
}

/**
 * Apply formatting rules to a sheet.
 */
function formatSheet(sheet, schema, rowCount) {
  formatMasterSheet(sheet.getName(), schema, rowCount);
  // apply Movements conditional rules if applicable
  if (sheet.getName() === 'Movements') {
    applyConditionalFormatting(sheet, movementsConditionalRules());
  }
}

function applyAlternatingColors(sheet, rowCount, colCount, altColor) {
  if (rowCount <= 1) return;
  altColor = altColor || COLORS.ALT_ROW;
  for (var r = 2; r <= rowCount; r++) {
    sheet.getRange(r, 1, 1, colCount).setBackground(r % 2 === 0 ? altColor : COLORS.WHITE);
  }
}

function refreshAllReadableSheets() {

  refreshDepartmentSummary();

}

function refreshCurrentView() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var mov = readSheetData('Movements');
  var posts = readSheetData('Posts');
  var col = readSheetData('Colleagues');

  var today = todayKey();
  var occupancy = computeOccupancy(mov, posts, today);

  var rows = [];
  for (var i = 0; i < posts.length; i++) {
    var p = posts[i];
    var key = (p.post || '') + '||' + (p.dept || '');
    var occ = occupancy[key] || {};
    rows.push([
      p.post || '',
      p.dept || '',
      occ.holder || '',
      lookupNickname(col, occ.holder),
      p.status === 'deleted' ? 'Obsolete' : (occ.status || 'Vacant'),
      occ.since || '',
      occ.pn || '',
      occ.holderType || '',
      occ.onAttachment || '',
      occ.futureIncoming || '',
      (p.roleNote || '') + (lookupPersonNote(col, occ.holder) ? '  |  ' + lookupPersonNote(col, occ.holder) : '')
    ]);
  }
  rows.sort(function(a, b) { return (a[1] || '').localeCompare(b[1] || '') || (a[0] || '').localeCompare(b[0] || ''); });

  writeReadableSheet('Current View',
    ['Post','Department','Holder','Nickname','Status','Since','PN No.','Holder Type','On Attachment','Future Incoming','Notes'],
    rows, [180,160,190,130,100,105,95,110,100,160,280], currentViewConditionalRules());
}

function refreshFutureMovements() {
  var mov = readSheetData('Movements');
  var today = todayKey();
  var rows = [];
  for (var i = 0; i < mov.length; i++) {
    var m = mov[i];
    var dk = m.dateKey || '';
    if (!dk || dk <= today) continue;
    rows.push([
      m.name || '',
      m.toPost || '',
      m.toDept || '',
      m.effDate || '',
      m.pn || '',
      m.remarkType || '',
      daysBetweenDK(today, dk)
    ]);
  }
  rows.sort(function(a, b) { return (a[6] || 99999) - (b[6] || 99999); });

  writeReadableSheet('Future Movements',
    ['Person','To Post','To Dept','Effective Date','PN No.','Remark Type','Days Until'],
    rows, [190,180,160,110,95,145,85], futureMovementsConditionalRules());
}

function refreshNoticeReview() {
  var not = readSheetData('Notices');
  var rows = not.map(function(n) {
    return [
      n.pn || '', n.year || '', n.status || '', n.moveCount || 0,
      n.supersededBy || '', n.reviewStatus || '', n.sourceFile || ''
    ];
  });
  rows.sort(function(a, b) {
    var ya = parseInt(a[1]) || 0, yb = parseInt(b[1]) || 0;
    if (ya !== yb) return yb - ya;
    return (b[0] || '').localeCompare(a[0] || '');
  });

  writeReadableSheet('Notice Review',
    ['PN No.','Year','Status','Movements','Superseded By','Review Status','Source File'],
    rows, [100,65,105,90,120,115,220], noticeReviewConditionalRules());
}

function refreshDepartmentSummary() {
  var posts = readSheetData('Posts');
  var mov = readSheetData('Movements');
  var occupancy = computeOccupancy(mov, posts, todayKey());

  var depts = {};
  for (var i = 0; i < posts.length; i++) {
    var p = posts[i];
    var d = p.dept || '(Unknown)';
    if (!depts[d]) depts[d] = { total:0, occupied:0, vacant:0, acting:0, future:0, obsolete:0 };
    depts[d].total++;
    if (p.status === 'deleted') { depts[d].obsolete++; continue; }
    var key = (p.post || '') + '||' + (p.dept || '');
    var occ = occupancy[key] || {};
    if (occ.status === 'Occupied') depts[d].occupied++;
    else if (occ.status === 'Acting') depts[d].acting++;
    else depts[d].vacant++;
    if (occ.futureIncoming) depts[d].future++;
  }

  var rows = Object.keys(depts).sort().map(function(d) {
    var c = depts[d];
    return [d, c.total, c.occupied, c.vacant, c.acting, c.future, c.obsolete];
  });
  rows.sort(function(a, b) { return (b[1] || 0) - (a[1] || 0); });

  writeReadableSheet('Department Summary',
    ['Department','Total Posts','Occupied','Vacant','Acting','Future Incoming','Obsolete'],
    rows, [200,95,90,80,80,100,85]);
}

function refreshExceptions() {
  var mov = readSheetData('Movements');
  var rows = [];
  for (var i = 0; i < mov.length; i++) {
    var m = mov[i];
    var issues = m.valIssues || '';
    var needsReview = m.reviewStatus === 'flagged' || m.needsReview === 'TRUE';
    var valStatus = m.valStatus || 'ok';
    if (issues || needsReview || valStatus !== 'ok') {
      rows.push([
        m.movId || '', m.name || '', issues, m.effDate || '',
        m.pn || '', needsReview ? 'Yes' : '', valStatus
      ]);
    }
  }
  rows.sort(function(a, b) {
    var order = { error:0, warning:1, ok:2 };
    return ((order[a[6]] || 9) - (order[b[6]] || 9)) || (b[3] || '').localeCompare(a[3] || '');
  });

  writeReadableSheet('Exceptions',
    ['Movement ID','Person','Issue','Date','PN No.','Needs Review','Validation'],
    rows, [100,190,320,105,95,105,105], exceptionsConditionalRules());
}

// ---- readable sheet writer ------------------------------------------------

function writeReadableSheet(name, headers, data, widths, condRules) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  sheet.clear();

  if (headers) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
  if (data && data.length > 0) {
    sheet.getRange(2, 1, data.length, data[0].length).setValues(data);
  }
  if (widths) {
    for (var w = 0; w < widths.length; w++) {
      if (widths[w] > 0) sheet.setColumnWidth(w + 1, widths[w]);
    }
  }
  sheet.setFrozenRows(1);
  if (sheet.getLastRow() > 1) {
    applyAlternatingColors(sheet, sheet.getLastRow(), headers.length, COLORS.ALT_ROW_READABLE);
  }
  // header style
  sheet.getRange(1, 1, 1, headers.length)
    .setFontWeight('bold')
    .setBackground(COLORS.HEADER_BG)
    .setBorder(true, true, true, true, true, true, COLORS.BORDER, SpreadsheetApp.BorderStyle.SOLID);
  // conditional formatting
  if (condRules) {
    applyConditionalFormatting(sheet, condRules);
  }
  // filter
  try {
    if (sheet.getLastRow() > 1) {
      sheet.getRange(1, 1, sheet.getLastRow(), headers.length).createFilter();
    }
  } catch(e) {}
}

// ---- master sheet reader (returns array of objects) -----------------------

function readSheetData(name) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sheet || sheet.getLastRow() <= 1) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data[0];

  // column index lookup
  var idx = {};
  for (var c = 0; c < headers.length; c++) idx[headers[c]] = c;

  var result = [];
  for (var r = 1; r < data.length; r++) {
    var row = data[r];
    result.push({
      // movements
      movId:       row[idx['Movement ID']],
      name:        row[idx['Name']],
      fromPost:    row[idx['From Post']],
      fromDept:    row[idx['From Dept']],
      toPost:      row[idx['To Post']],
      toDept:      row[idx['To Dept']],
      effDate:     row[idx['Effective Date']],
      pn:          row[idx['PN No.']],
      remarkType:  row[idx['Remark Type']],
      remarkFlags: row[idx['Remark Flags']],
      rawRemark:   row[idx['Raw Remark']],
      personId:    row[idx['Person ID']],
      noticeId:    row[idx['Notice ID']],
      valStatus:   row[idx['Validation Status']],
      valIssues:   row[idx['Validation Issues']],
      superseded:  row[idx['Superseded']],
      cancelled:   row[idx['Cancelled']],
      actingFlag:  row[idx['Acting Flag']],
      reviewStatus:row[idx['Review Status']],
      needsReview: row[idx['Review Status']],   // approximate
      dateKey:     parseDateKeyLocal(row[idx['Effective Date']] || ''),
      // notices
      year:        row[idx['Issued Year']],
      status:      row[idx['Status']],
      moveCount:   row[idx['Movement Count']],
      supersededBy:row[idx['Superseded By']],
      sourceFile:  row[idx['Source File']],
      // colleagues
      nickname:    row[idx['Nickname']],
      notes:       row[idx['Notes']],
      // posts
      post:        row[idx['Post Title']],
      dept:        row[idx['Department']],
      rankGroup:   row[idx['Rank Group']],
      roleNote:    row[idx['Role Note']],
      holder:      row[idx['Current Holder']],
      since:       row[idx['Current Since']]
    });
  }
  return result;
}

// ---- occupy engine (used by readable sheet generation) --------------------

function computeOccupancy(mov, posts, today) {
  var state = {};
  var sorted = mov.slice().sort(function(a, b) {
    if ((a.dateKey || '') !== (b.dateKey || '')) return (a.dateKey || '').localeCompare(b.dateKey || '');
    return (a.pn || '').localeCompare(b.pn || '');
  });

  for (var i = 0; i < sorted.length; i++) {
    var m = sorted[i];
    if (m.superseded === 'TRUE' || m.cancelled === 'TRUE') continue;
    var dk = m.dateKey || '';
    var isPast = dk <= today;

    var fromKey = (m.fromPost || '') + '||' + (m.fromDept || '');
    if (fromKey !== '||' && isPast) {
      if (state[fromKey] && state[fromKey].holder === (m.name || '')) {
        state[fromKey] = { status: 'Vacant', holder: '', since: '', pn: '', holderType: '', onAttachment: '', futureIncoming: '' };
      }
    }

    var toKey = (m.toPost || '') + '||' + (m.toDept || '');
    if (toKey === '||') continue;
    var isFuture = dk > today;

    if (isFuture) {
      if (!state[toKey]) state[toKey] = { status: 'Occupied', holder: '', since: '', pn: '', holderType: '', onAttachment: '', futureIncoming: '' };
      state[toKey].futureIncoming = m.name || '';
      continue;
    }

    var rt = (m.remarkType || '').toLowerCase();
    var isAct  = rt === 'acting' || rt === 'continue-acting';
    var isCease = rt === 'cease-acting';
    var isOut  = rt === 'transfer-out' || rt === 'no-pay-leave';
    var isAtt  = rt === 'attachment';

    if (isCease || isOut) {
      if (state[toKey] && state[toKey].holder === (m.name || '')) {
        state[toKey] = { status: 'Vacant', holder: '', since: '', pn: '', holderType: '', onAttachment: '', futureIncoming: (state[toKey] || {}).futureIncoming || '' };
      }
    } else if (isAct) {
      state[toKey] = { status: 'Acting', holder: m.name || '', since: m.effDate || '', pn: m.pn || '', holderType: 'Acting', onAttachment: '', futureIncoming: (state[toKey] || {}).futureIncoming || '' };
    } else if (isAtt) {
      state[toKey] = { status: 'Occupied', holder: m.name || '', since: m.effDate || '', pn: m.pn || '', holderType: 'Substantive', onAttachment: 'Yes', futureIncoming: (state[toKey] || {}).futureIncoming || '' };
    } else {
      state[toKey] = { status: 'Occupied', holder: m.name || '', since: m.effDate || '', pn: m.pn || '', holderType: 'Substantive', onAttachment: '', futureIncoming: (state[toKey] || {}).futureIncoming || '' };
    }
  }
  return state;
}

function lookupNickname(col, name) {
  if (!name) return '';
  for (var i = 0; i < col.length; i++) {
    if ((col[i].name || '') === name) return col[i].nickname || '';
  }
  return '';
}

function lookupPersonNote(col, name) {
  if (!name) return '';
  for (var i = 0; i < col.length; i++) {
    if ((col[i].name || '') === name) return col[i].notes || '';
  }
  return '';
}

// ---- date helpers ---------------------------------------------------------

function todayKey() {
  var d = new Date();
  return d.getFullYear() + String(d.getMonth() + 1).padStart(2, '0') + String(d.getDate()).padStart(2, '0');
}

function parseDateKeyLocal(dateStr) {
  var raw = String(dateStr || '').trim();
  var m = raw.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})$/);
  if (m) return m[3] + m[2].padStart(2, '0') + m[1].padStart(2, '0');
  return raw;
}

function daysBetweenDK(fromKey, toKey) {
  if (fromKey.length !== 8 || toKey.length !== 8) return '';
  var fy = parseInt(fromKey.substring(0,4)), fm = parseInt(fromKey.substring(4,6)), fd = parseInt(fromKey.substring(6,8));
  var ty = parseInt(toKey.substring(0,4)),   tm = parseInt(toKey.substring(4,6)),   td = parseInt(toKey.substring(6,8));
  return Math.round((new Date(ty, tm-1, td) - new Date(fy, fm-1, fd)) / 86400000);
}

// ---- ID generation --------------------------------------------------------

function hashIdLocal(str) {
  var h = 5381;
  for (var i = 0; i < str.length; i++) h = ((h << 5) + h + str.charCodeAt(i)) | 0;
  if (h < 0) h = -h;
  return ('000000' + h.toString(36)).slice(-6);
}

// ---- row normalisation ----------------------------------------------------


function toSheetEffectiveDate(value) {
  var iso = normaliseEffectiveDate(value);
  var p = iso.split('-');
  return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
}
function sheetDateToIso(value) {
  if (!(value instanceof Date) || isNaN(value)) return '';
  return value.getFullYear() + '-' + String(value.getMonth() + 1).padStart(2, '0') + '-' + String(value.getDate()).padStart(2, '0');
}

function rowVal(row, col) {
  if (row[col] === undefined || row[col] === null) return '';
  var v = row[col];
  return v instanceof Date ? formatDateLocal(v) : String(v).trim();
}

function formatDateLocal(d) {
  return d.getDate() + '.' + (d.getMonth() + 1) + '.' + d.getFullYear();
}

// ---- POST: master sheet writers -------------------------------------------

function writeMovementsFromPayload(rows) {
  if (!rows || rows.length < 2) return;
  var hdr = rows[0];
  var map = buildColMap(hdr);
  var out = [];
  for (var r = 1; r < rows.length; r++) {
    var row = rows[r];
    var name     = rowVal(row, map['姓名 (Name)'])     || rowVal(row, map['Name'])     || '';
    var fromPost = rowVal(row, map['原職 (From Post)']) || rowVal(row, map['From Post']) || '';
    var fromDept = rowVal(row, map['原部門 (From Dept)'])|| rowVal(row, map['From Dept'])|| '';
    var toPost   = rowVal(row, map['現職 (To Post)'])   || rowVal(row, map['To Post'])   || '';
    var toDept   = rowVal(row, map['現部門 (To Dept)']) || rowVal(row, map['To Dept'])   || '';
    var effDate  = rowVal(row, map['生效日期'])         || rowVal(row, map['Effective Date']) || '';
    var pn       = rowVal(row, map['PN No.'])           || rowVal(row, map['PN No.'])    || '';
    var remark   = rowVal(row, map['備註 (Remark)'])    || rowVal(row, map['Remark'])     || '';
    if (!name) continue;

    var personId   = hashIdLocal(name.toLowerCase().replace(/\s+/g, ''));
    var fromPostId = hashIdLocal(fromPost.toLowerCase());
    var fromDeptId = hashIdLocal(fromDept.toLowerCase());
    var toPostId   = hashIdLocal(toPost.toLowerCase());
    var toDeptId   = hashIdLocal(toDept.toLowerCase());
    var noticeId   = hashIdLocal(pn || '');
    var movId      = hashIdLocal(personId + toPostId + toDeptId + parseDateKeyLocal(effDate) + noticeId);

    out.push([
      movId, name, fromPost, fromDept, toPost, toDept, effDate, pn,
      '', '', remark,
      personId, fromPostId, fromDeptId, toPostId, toDeptId, noticeId,
      'gdrive-sync', 'unreviewed', 'ok', '',
      'FALSE', 'FALSE', 'FALSE',
      new Date().toISOString(), new Date().toISOString()
    ]);
  }
  writeSheetWithHeaders('Movements', SCHEMAS.Movements, out);
  syncNoticesFromMovements(rows, buildColMap(hdr));
}

function writeColleaguesFromPayload(rows) {
  if (!rows || rows.length < 2) return;
  var hdr = rows[0];
  var map = buildColMap(hdr);
  var out = [];
  var seen = {};
  for (var r = 1; r < rows.length; r++) {
    var row = rows[r];
    var name = rowVal(row, map['姓名 (Name)']) || rowVal(row, map['Name']) || '';
    if (!name || seen[name]) continue;
    seen[name] = true;
    var nick  = rowVal(row, map['暱稱 (Nickname)'])   || rowVal(row, map['Nickname'])   || '';
    var notes = rowVal(row, map['人物備註'])          || rowVal(row, map['Person Remark']) || '';
    var chunks = [];
    for (var ck = 0; ck < hdr.length; ck++) {
      if ((hdr[ck] || '').indexOf('人物相片資料_') === 0) chunks.push(row[ck] || '');
    }
    out.push([
      hashIdLocal(name.toLowerCase().replace(/\s+/g, '')),
      name, nick, 'Unknown', notes, '', '',
      chunks[0]||'', chunks[1]||'', chunks[2]||'', chunks[3]||'',
      chunks[4]||'', chunks[5]||'', chunks[6]||'', chunks[7]||'',
      new Date().toISOString(), new Date().toISOString()
    ]);
  }
  writeSheetWithHeaders('Colleagues', SCHEMAS.Colleagues, out);
}

function writePostsFromPayload(rows) {
  if (!rows || rows.length < 2) return;
  var hdr = rows[0];
  var map = buildColMap(hdr);
  var out = [];
  for (var r = 1; r < rows.length; r++) {
    var row  = rows[r];
    var post   = rowVal(row, map['職位 (Post)'])       || rowVal(row, map['Post'])       || '';
    var dept   = rowVal(row, map['部門 (Department)']) || rowVal(row, map['Department']) || '';
    var status = rowVal(row, map['狀態 (Status)'])     || rowVal(row, map['Status'])     || 'active';
    var holder = rowVal(row, map['現任 (Holder)'])     || rowVal(row, map['Holder'])     || '';
    var rnote  = rowVal(row, map['崗位備註 (Remark)']) || rowVal(row, map['Remark'])     || '';
    out.push([
      hashIdLocal(post + '||' + dept),
      post, dept, classifyRankLocal(post), status, rnote, holder, '',
      new Date().toISOString(), new Date().toISOString()
    ]);
  }
  writeSheetWithHeaders('Posts', SCHEMAS.Posts, out);
}

function syncNoticesFromMovements(rows, map) {
  var counts = {};
  for (var r = 1; r < rows.length; r++) {
    var pn = rowVal(rows[r], map['PN No.']) || rowVal(rows[r], map['PN No.']) || '';
    if (!pn) continue;
    counts[pn] = (counts[pn] || 0) + 1;
  }
  var out = [];
  var keys = Object.keys(counts).sort();
  for (var i = 0; i < keys.length; i++) {
    var pn     = keys[i];
    var m      = String(pn).match(/(\d+)\s*\/\s*(\d{4})/);
    var serial = m ? String(m[1]).padStart(4, '0') + '/' + m[2] : '';
    out.push([
      hashIdLocal(serial || pn), pn, serial,
      m ? parseInt(m[1]) : 0, m ? parseInt(m[2]) : 0,
      'active', 'unreviewed', '', counts[pn], '',
      new Date().toISOString(), new Date().toISOString()
    ]);
  }
  writeSheetWithHeaders('Notices', SCHEMAS.Notices, out);
}

function buildColMap(headers) {
  var map = {};
  for (var c = 0; c < headers.length; c++) map[headers[c]] = c;
  return map;
}

var RANK_ORDER = ['C', 'PSO', 'CSO', 'SSO', 'SO', 'ASO'];

function classifyRankLocal(title) {
  var s = (title || '').trim();
  for (var i = 0; i < RANK_ORDER.length; i++) {
    if (s === RANK_ORDER[i] || s.indexOf(RANK_ORDER[i] + '(') === 0 || s.indexOf(RANK_ORDER[i] + ' ') === 0) return RANK_ORDER[i];
  }
  return 'OTHER';
}

/**
 * Project canonical Movements data back into legacy Records flat-array format.
 * This allows old front-end clients to read data written by new clients.
 *
 * @param {Sheet} movSheet — the Movements sheet
 * @return {Array<Array>} 2D array: [headerRow, ...dataRows]
 */
function projectRecordsFromMovements(movSheet) {
  if (!movSheet || movSheet.getLastRow() <= 1) {
    return [['姓名 (Name)','原職 (From Post)','原部門 (From Dept)','原崗位備註',
             '現職 (To Post)','現部門 (To Dept)','現崗位備註','生效日期',
             '備註 (Remark)','PN No.']];
  }
  var raw = movSheet.getDataRange().getValues();
  var result = [['姓名 (Name)','原職 (From Post)','原部門 (From Dept)','原崗位備註',
                 '現職 (To Post)','現部門 (To Dept)','現崗位備註','生效日期',
                 '備註 (Remark)','PN No.']];

  for (var r = 1; r < raw.length; r++) {
    var row = raw[r];
    var name     = cellStr(row[1]);   // B = Name
    var fromPost = cellStr(row[2]);   // C = From Post
    var fromDept = cellStr(row[3]);   // D = From Dept
    var toPost   = cellStr(row[4]);   // E = To Post
    var toDept   = cellStr(row[5]);   // F = To Dept
    var effDate  = cellStr(row[6]);   // G = Effective Date
    var remark   = cellStr(row[10]);  // K = Raw Remark
    var pn       = cellStr(row[7]);   // H = PN No.
    if (!name) continue;

    result.push([name, fromPost, fromDept, '', toPost, toDept, '', effDate, remark, pn]);
  }

  normalizePnAsText(result);
  return result;
}

/**
 * Build compatibility warnings metadata for legacy-mode responses.
 */
function buildCompatWarnings(legacySheetsWritten) {
  var warnings = [];
  if (compatWarningsEnabled()) {
    warnings.push({
      code: 'LEGACY_FORMAT',
      severity: 'warning',
      message: getDeprecationNotice()
    });
    warnings.push({
      code: 'SCHEMA_VERSION',
      severity: 'info',
      message: 'Current workbook schema: v' + REQUIRED_SCHEMA_VERSION + '. Upgrade to canonical Movements-based format for full feature support.'
    });
    if (legacySheetsWritten && legacySheetsWritten.length > 0) {
      warnings.push({
        code: 'LEGACY_WRITE',
        severity: 'info',
        message: 'Legacy sheets written: ' + legacySheetsWritten.join(', ') + '. Data was migrated to canonical masters.'
      });
    }
  }
  return warnings;
}

/**
 * Determine which format to use for GET response based on ?mode param or Config.
 */
function resolveResponseFormat(requestedMode) {
  if (requestedMode === 'canonical') return 'canonical';
  if (requestedMode === 'legacy') return 'legacy';
  if (requestedMode === 'full') return 'full';       // legacy + readables
  // fall back to Config-driven behaviour
  var configMode = getMigrationMode();
  if (configMode === 'canonical') return 'canonical';
  if (configMode === 'dual') return 'full';           // dual = send everything
  return 'legacy';                                     // legacy-only
}

/**
 * Build the legacy-format GET response.
 */
function buildLegacyResponse(ss) {
  var result = {};
  var mov = ss.getSheetByName('Movements');
  var col = ss.getSheetByName('Colleagues');
  var posts = ss.getSheetByName('Posts');

  result.Records    = projectRecordsFromMovements(mov);
  result.Colleagues = projectColleaguesFromColleagues(col);
  result.Posts      = projectPostsFromPosts(posts);
  result._compatWarnings = buildCompatWarnings([]);
  result._migrationHint = 'Use ?mode=canonical for Movements-based format. Legacy format is deprecated.';
  return result;
}

/**
 * Build the canonical GET response (raw sheet data).
 */
function buildCanonicalResponse(ss, includeReadables) {
  var result = {};
  var masterNames = ['Movements', 'Notices', 'Colleagues', 'Posts', 'Config'];
  for (var i = 0; i < masterNames.length; i++) {
    var sheet = ss.getSheetByName(masterNames[i]);
    result[masterNames[i]] = sheet && sheet.getLastRow() > 0 ? sheet.getDataRange().getValues() : [];
  }
  result._schemaVersion = REQUIRED_SCHEMA_VERSION;
  result._migrationMode = getMigrationMode();

  if (includeReadables) {
    var readableNames = ['Current View', 'Future Movements', 'Notice Review', 'Department Summary', 'Exceptions'];
    for (var j = 0; j < readableNames.length; j++) {
      result[readableNames[j].replace(/\s/g, '')] = readRawSheet(readableNames[j]);
    }
  }
  return result;
}

/**
 * Detect whether a POST payload contains legacy sheet names.
 * @return {Array} list of legacy sheet names found, or empty array.
 */
function detectLegacyPayload(sheetNames) {
  var legacy = [];
  for (var i = 0; i < sheetNames.length; i++) {
    if (sheetNames[i] === 'Records' || sheetNames[i] === 'Colleagues' || sheetNames[i] === 'Posts') {
      legacy.push(sheetNames[i]);
    }
  }
  return legacy;
}

/**
 * Set the migration mode in Config. Useful for one-shot migration.
 */
function setMigrationMode(mode) {
  if (['legacy', 'dual', 'canonical'].indexOf(mode) === -1) {
    throw new Error('Invalid migration mode: ' + mode + '. Use legacy, dual, or canonical.');
  }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config');
  var data = sheet.getDataRange().getValues();
  for (var r = 1; r < data.length; r++) {
    if (data[r][0] === MIGRATION_MODE_KEY) {
      sheet.getRange(r + 1, 2).setValue(mode);
      sheet.getRange(r + 1, 4).setValue(new Date().toISOString());
      return;
    }
  }
  // key not found, append
  sheet.appendRow([MIGRATION_MODE_KEY, mode, 'Migration mode: legacy | dual | canonical', new Date().toISOString()]);
}

/**
 * One-shot migration: lock into canonical mode and remove legacy compat.
 * Run once all clients have been upgraded.
 */
function finaliseMigration() {
  setMigrationMode('canonical');
  SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config')
    .getRange(2, 2).setValue('false');  // COMPAT_WARNINGS = false (row 6)
  appendAuditLog('migrate', 'workbook', '', 'Migration finalised: mode=canonical, compat warnings disabled', '');
  Logger.log('Migration finalised. Mode set to canonical.');
}

function cellStr(val) {
  if (val === undefined || val === null) return '';
  return val instanceof Date ? formatDateLocal(val) : String(val).trim();
}

function doGet(e) {
  try {
    // ---- diagnostic ping (no auth, no spreadsheet required) ----
    if (e && e.parameter && e.parameter.ping === '1') {
      return jsonResp({
        success: true,
        ping: 'ok',
        time: new Date().toISOString(),
        version: '2.2',
        schemaVersion: REQUIRED_SCHEMA_VERSION,
        scriptType: typeof SpreadsheetApp.getActiveSpreadsheet() === 'object' ? 'bound' : 'standalone'
      });
    }

    // ---- full diagnostics (needs auth) ----
    var pwd = (e && e.parameter && e.parameter.pwd) || '';
    var mode = (e && e.parameter && e.parameter.mode) || '';

    if (mode === 'diag') {
      if (!validatePassword(pwd)) return errorResp('密碼錯誤');
      return jsonResp(runFullDiagnostics());
    }

    if (!validatePassword(pwd)) return errorResp('密碼錯誤');

    ensureAllMasterSheets();
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return errorResp('找不到 Google Sheets 文件。請確保 Apps Script 已綁定至試算表。');

    var format = resolveResponseFormat(mode);
    var result;

    if (format === 'canonical') {
      result = buildCanonicalResponse(ss, false);
    } else if (format === 'full') {
      result = buildLegacyResponse(ss);
      result.CurrentView       = readRawSheet('Current View');
      result.FutureMovements   = readRawSheet('Future Movements');
      result.NoticeReview      = readRawSheet('Notice Review');
      result.DepartmentSummary = readRawSheet('Department Summary');
      result.Exceptions        = readRawSheet('Exceptions');
      result._canonicalAvailable = true;
    } else {
      result = buildLegacyResponse(ss);
    }

    return jsonResp(result);
  } catch (err) {
    return errorResp('伺服器錯誤: ' + err.toString());
  }
}

// ---- POST handler ---------------------------------------------------------

function doPost(e) {
  try {
    var raw = e && e.postData && e.postData.contents;
    if (!raw) return errorResp('請求內容為空');

    var payload;
    try { payload = JSON.parse(raw); }
    catch (err) { return errorResp('無法解析請求內容'); }

    var pwd = payload.pwd !== undefined ? String(payload.pwd) : '';
    if (!validatePassword(pwd)) return errorResp('密碼錯誤');

    var schemaErr = validateSchemaVersion(payload);
    if (schemaErr) return errorResp(schemaErr);

    var action = (payload.action || '').toString();
    if (ALLOWED_ACTIONS.indexOf(action) === -1) return errorResp('不支援的操作: ' + action);

    var sheets = payload.sheets || {};
    var names = Object.keys(sheets);
    if (names.length === 0) return errorResp('缺少 sheets 資料');

    // validate sheet names
    for (var i = 0; i < names.length; i++) {
      if (ALLOWED_WRITE_SHEETS.indexOf(names[i]) === -1) {
        return errorResp('不允許的工作表名稱: ' + names[i]);
      }
    }

    ensureAllMasterSheets();

    // detect legacy payload
    var legacyWritten = detectLegacyPayload(names);

    // process each sheet
    for (var j = 0; j < names.length; j++) {
      var name = names[j];
      if (name === 'Records')      writeMovementsFromPayload(sheets[name]);
      else if (name === 'Colleagues') writeColleaguesFromPayload(sheets[name]);
      else if (name === 'Posts')      writePostsFromPayload(sheets[name]);
      else if (SCHEMAS[name])         writeSheetWithHeaders(name, SCHEMAS[name], sheets[name]);
    }

    // refresh readable sheets
    refreshAllReadableSheets();

    // audit
    appendAuditLog(action, 'workbook', '', action + ' completed',
      JSON.stringify({ sheets: names, legacyDetected: legacyWritten.length > 0 }).substring(0, 500));

    var response = {
      success: true,
      schemaVersion: REQUIRED_SCHEMA_VERSION,
      _migrationMode: getMigrationMode()
    };

    if (legacyWritten.length > 0) {
      response._compatWarnings = buildCompatWarnings(legacyWritten);
      response._migrationHint = getDeprecationNotice();
    }

    return jsonResp(response);
  } catch (err) {
    return errorResp('伺服器錯誤: ' + err.toString());
  }
}

// ---- utility --------------------------------------------------------------

function readRawSheet(name) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sheet || sheet.getLastRow() === 0) return [];
  return sheet.getDataRange().getValues();
}

function normalizePnAsText(rows) {
  for (var r = 1; r < rows.length; r++) {
    var cell = rows[r][9];  // column index 9 = PN No. in legacy Records format
    if (typeof cell === 'string' && /^\d{4}-\d{2}-\d{2}T/.test(cell)) {
      var d = new Date(cell);
      rows[r][9] = (d.getUTCMonth() + 1) + '/' + d.getUTCFullYear();
    }
  }
}

// ---- full diagnostics (callable via ?mode=diag) ---------------------------

function runFullDiagnostics() {
  var diag = { success: true, time: new Date().toISOString() };

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    diag.boundToSpreadsheet = !!ss;
    if (!ss) {
      diag.error = 'Apps Script is NOT bound to a spreadsheet. Use Tools > Script Editor from within a Google Sheet.';
      return diag;
    }

    diag.spreadsheetId   = ss.getId();
    diag.spreadsheetName = ss.getName();
    diag.spreadsheetUrl  = ss.getUrl();

    // sheet inventory
    var allSheets = ss.getSheets();
    diag.sheetCount = allSheets.length;
    diag.sheets = [];
    for (var i = 0; i < allSheets.length; i++) {
      var s = allSheets[i];
      diag.sheets.push({
        name: s.getName(),
        rows: s.getLastRow(),
        cols: s.getLastColumn(),
        frozen: s.getFrozenRows()
      });
    }

    // config values
    diag.config = {};
    var cfgSheet = ss.getSheetByName('Config');
    if (cfgSheet && cfgSheet.getLastRow() > 1) {
      var cfgData = cfgSheet.getDataRange().getValues();
      for (var r = 1; r < cfgData.length; r++) {
        diag.config[cfgData[r][0]] = cfgData[r][1];
      }
    }

    // password check
    diag.passwordConfigured = !!getConfigValue('APP_PASSWORD');
    diag.migrationMode = getMigrationMode();
    diag.schemaVersion = REQUIRED_SCHEMA_VERSION;

    // master sheet counts
    var masterNames = ['Movements', 'Notices', 'Colleagues', 'Posts', 'Config', 'Audit Log'];
    diag.masterCounts = {};
    for (var m = 0; m < masterNames.length; m++) {
      var ms = ss.getSheetByName(masterNames[m]);
      diag.masterCounts[masterNames[m]] = ms ? ms.getLastRow() - 1 : -1;  // -1 = missing
    }

    // record count (Movements data rows)
    var movSheet = ss.getSheetByName('Movements');
    diag.recordCount = movSheet && movSheet.getLastRow() > 1 ? movSheet.getLastRow() - 1 : 0;

  } catch (err) {
    diag.error = err.toString();
  }

  return diag;
}

/**
 * Apply full formatting to a master sheet.
 *   freeze row 1 · hide tech cols · alternating colours · PN/date as text · filter
 */
function formatMasterSheet(name, schema, rowCount) {
  if (name === 'Movements' && rowCount > 1) {
    SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name).getRange(2, 7, rowCount - 1, 1).setNumberFormat('dd.MM.yyyy');
  }
  if (!rowCount || rowCount < 2) rowCount = 2;
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sheet) return;

  freezeAndFilter(sheet, rowCount, schema.headers.length);
  hideTechnicalColumns(sheet, schema.hidden);
  applyAlternatingColors(sheet, rowCount, schema.headers.length, COLORS.ALT_ROW);

  // text-format columns that must never be auto-converted
  for (var t = 0; t < (schema.textCols || []).length; t++) {
    var col = schema.textCols[t];
    if (rowCount >= 2) sheet.getRange(2, col, Math.max(rowCount - 1, 1), 1).setNumberFormat('@');
  }

  // ID columns: force text to prevent numeric interpretation
  applyTextFormatToIds(sheet, schema, rowCount);

  // column widths
  for (var w = 0; w < (schema.widths || []).length; w++) {
    if (schema.widths[w] > 0) sheet.setColumnWidth(w + 1, schema.widths[w]);
  }

  // header style
  if (rowCount > 0) {
    sheet.getRange(1, 1, 1, schema.headers.length)
      .setFontWeight('bold')
      .setBackground(COLORS.HEADER_BG)
      .setBorder(true, true, true, true, true, true, COLORS.BORDER, SpreadsheetApp.BorderStyle.SOLID);
  }
}

/**
 * Apply full formatting to a readable sheet with conditional rules.
 */
function formatReadableSheet(name, headers, data, widths, condRules) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sheet) return;
  var rowCount = sheet.getLastRow();
  if (rowCount < 2) rowCount = 2;

  freezeAndFilter(sheet, rowCount, (headers || data[0] || []).length);
  applyAlternatingColors(sheet, rowCount, (headers || []).length || (data && data[0] ? data[0].length : 6), COLORS.ALT_ROW_READABLE);

  if (widths) {
    for (var w = 0; w < widths.length; w++) {
      if (widths[w] > 0) sheet.setColumnWidth(w + 1, widths[w]);
    }
  }

  // header
  var colCount = (headers || data[0] || []).length;
  sheet.getRange(1, 1, 1, colCount)
    .setFontWeight('bold')
    .setBackground(COLORS.HEADER_BG)
    .setBorder(true, true, true, true, true, true, COLORS.BORDER, SpreadsheetApp.BorderStyle.SOLID);

  // conditional formatting
  if (condRules) {
    applyConditionalFormatting(sheet, condRules);
  }
}

// ---- hideTechnicalColumns --------------------------------------------------

function hideTechnicalColumns(sheet, hiddenCols) {
  if (!hiddenCols || hiddenCols.length === 0) return;
  var lastCol = sheet.getLastColumn();
  for (var i = 0; i < hiddenCols.length; i++) {
    var col = hiddenCols[i];
    if (col <= lastCol) {
      try { sheet.hideColumns(col); } catch(e) {}
    }
  }
}

// ---- applyTextFormatToIds --------------------------------------------------

function applyTextFormatToIds(sheet, schema, rowCount) {
  // Any column whose header contains "ID" or "Id" gets @ format
  if (!schema || !schema.headers) return;
  for (var c = 0; c < schema.headers.length; c++) {
    if (/\bID\b/i.test(schema.headers[c] || '')) {
      var colIdx = c + 1;
      if (rowCount >= 2) sheet.getRange(2, colIdx, rowCount - 1, 1).setNumberFormat('@');
    }
  }
}

// ---- freezeAndFilter -------------------------------------------------------

function freezeAndFilter(sheet, rowCount, colCount) {
  sheet.setFrozenRows(1);
  // Create a filter on the data range if supported (doesn't work on all GAS runtimes)
  try {
    if (rowCount > 1 && colCount > 0) {
      sheet.getRange(1, 1, rowCount, colCount).createFilter();
    }
  } catch(e) {}
}

// ---- autoResizeSafe --------------------------------------------------------

function autoResizeSafe(sheet, minWidth, maxWidth) {
  minWidth = minWidth || 80;
  maxWidth = maxWidth || 400;
  var lastCol = sheet.getLastColumn();
  for (var c = 1; c <= lastCol; c++) {
    try { sheet.autoResizeColumn(c); } catch(e) {}
    var w = sheet.getColumnWidth(c);
    if (w < minWidth) sheet.setColumnWidth(c, minWidth);
    if (w > maxWidth) sheet.setColumnWidth(c, maxWidth);
  }
}

// ---- addHeaderNotes --------------------------------------------------------

/**
 * Insert an explanatory row just below the header row on a readable sheet.
 * The row uses smaller grey italic text.
 */
function addHeaderNotes(sheet, notes) {
  if (!notes || notes.length === 0) return;
  sheet.insertRowAfter(1);
  var range = sheet.getRange(2, 1, 1, notes.length);
  range.setValues([notes]);
  range.setFontSize(9).setFontStyle('italic').setFontColor(COLORS.GREY_TEXT);
  // Re-apply frozen row to include the note row
  sheet.setFrozenRows(2);
}

// ---- applyConditionalFormatting --------------------------------------------

/**
 * Apply conditional formatting rules to a sheet.
 *
 * @param {Sheet} sheet
 * @param {Array} rules — array of { col, test, bg, fg, bold }
 *   col:   1-indexed column number
 *   test:  custom formula as a string, e.g. '=A2="Vacant"'
 *   bg:    background colour hex
 *   fg:    text colour hex (optional)
 *   bold:  true/false (optional)
 */
function applyConditionalFormatting(sheet, rules) {
  if (!rules || rules.length === 0) return;
  var maxRows = sheet.getMaxRows();

  // clear ALL existing conditional formatting rules on this sheet first
  sheet.clearConditionalFormatRules();

  var newRules = [];
  for (var i = 0; i < rules.length; i++) {
    var r = rules[i];
    if (!r.test || !r.col) continue;

    var range = sheet.getRange(1, r.col, Math.max(maxRows, 2), 1);
    var rule = SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied(r.test)
      .setBackground(r.bg || '#FFFFFF')
      .setRanges([range])
      .build();
    newRules.push(rule);
  }
  if (newRules.length > 0) {
    sheet.setConditionalFormatRules(newRules);
  }
}

function migrateMovementDatesToRealDates() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Movements');
  if (!sheet || sheet.getLastRow() < 2) return;
  var range = sheet.getRange(2, 7, sheet.getLastRow() - 1, 1);
  var values = range.getValues();
  for (var i = 0; i < values.length; i++) if (values[i][0] !== '') values[i][0] = toSheetEffectiveDate(values[i][0]);
  range.setValues(values).setNumberFormat('dd.MM.yyyy');
}
